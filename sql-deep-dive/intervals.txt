1) Interval 

    select * 
    from orders 
    where purchaseDate <= now() - interval '30 days' 
    --> purchaseDate less than or equal to 30 days ago

    Note: INTERVAL '30 days' --> represents a duration of 30 days
        INTERVAL '1 year 2 months 3 days';
        INTERVAL '2 weeks ago';
        INTERVAL '1 year 3 hours 20 minutes';

2) Extracting 
    select extract(year from interval '5 years 20 months'); --> 6 years
